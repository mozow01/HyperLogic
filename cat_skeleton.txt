Class IsRightAdjoint {C D : SetoidCategory} (G : Functor C D) := mk_IsRightAdjoint {
  left_adj_obj : @obj D -> @obj C;

  unit : forall (Y : @obj D), @hom D Y (@fobj C D G (left_adj_obj Y));

  left_adj_mor : forall {Y : @obj D} {X : @obj C} (g : @hom D Y (@fobj C D G X)), 
    @hom C (left_adj_obj Y) X;

  left_adj_mor_proper : forall {Y : @obj D} {X : @obj C},
    Proper (@hom_equiv D Y (@fobj C D G X) ==> @hom_equiv C (left_adj_obj Y) X) left_adj_mor;

  lambek_1_dual : forall {Y : @obj D} {X : @obj C} (g : @hom D Y (@fobj C D G X)),
    @hom_equiv D Y (@fobj C D G X)
      (@compose D Y (@fobj C D G (left_adj_obj Y)) (@fobj C D G X) 
         (@fmap C D G (left_adj_obj Y) X (left_adj_mor g)) 
         (unit Y))
      g;

  lambek_2_dual : forall {Y : @obj D} {X : @obj C} (f : @hom C (left_adj_obj Y) X),
    @hom_equiv C (left_adj_obj Y) X
      (left_adj_mor 
        (@compose D Y (@fobj C D G (left_adj_obj Y)) (@fobj C D G X) 
           (@fmap C D G (left_adj_obj Y) X f) 
           (unit Y)))
      f
}.

Existing Instance left_adj_mor_proper.

Class IsLeftAdjoint {C D : SetoidCategory} (F : Functor D C) := mk_IsLeftAdjoint {
  right_adj_obj : @obj C -> @obj D;

  epsilon : forall (X : @obj C), @hom C (@fobj D C F (right_adj_obj X)) X;

  right_adj_mor : forall {Y : @obj D} {X : @obj C} (f : @hom C (@fobj D C F Y) X), 
    @hom D Y (right_adj_obj X);

  right_adj_mor_proper : forall {Y : @obj D} {X : @obj C},
    Proper (@hom_equiv C (@fobj D C F Y) X ==> @hom_equiv D Y (right_adj_obj X)) right_adj_mor;

  lambek_1 : forall {Y : @obj D} {X : @obj C} (f : @hom C (@fobj D C F Y) X),
    @hom_equiv C (@fobj D C F Y) X
      (@compose C (@fobj D C F Y) (@fobj D C F (right_adj_obj X)) X
         (epsilon X)
         (@fmap D C F Y (right_adj_obj X) (right_adj_mor f)))
      f;

  lambek_2 : forall {Y : @obj D} {X : @obj C} (g : @hom D Y (right_adj_obj X)),
    @hom_equiv D Y (right_adj_obj X)
      (right_adj_mor
        (@compose C (@fobj D C F Y) (@fobj D C F (right_adj_obj X)) X
           (epsilon X)
           (@fmap D C F Y (right_adj_obj X) g)))
      g
}.

Existing Instance right_adj_mor_proper.

(* If LA is a left adjoint, then the right component gives rise of a functor *)
Definition RightAdjointFunctor {C D : SetoidCategory} (F : Functor D C) (LA : IsLeftAdjoint F) 
  : Functor C D.
Proof. Admitted.

(* If RA (with G : C -> D) is a right adjoint structure, then it is constructed a functor F : D -> C *)
Definition LeftAdjointFunctor {C D : SetoidCategory} (G : Functor C D) (RA : IsRightAdjoint G)
  : Functor D C.
Proof. Admitted.

(* Itt a változás: {Ct : CartesianCategory C} hozzáadása *)
Class BiCartesianClosedCategory {C : SetoidCategory} {Ct : CartesianCategory C} (CC : CartesianClosedCategory C) := mk_BiCartesianClosedCategory {

  (* ========================== *)
  (* 1. INITIAL OBJECT          *)
  (* ========================== *)
  initial : @obj C;
  initial_map : forall {x}, hom initial x;

  (* Eta rule for Initial Object *)
  initial_map_unique : forall {x} (f : hom initial x), 
        f ≅ initial_map;

  (* ========================== *)
  (* 2. COPRODUCT               *)
  (* ========================== *)
  coprod : @obj C -> @obj C -> @obj C;
  (* Constructors (Injections) *)
  coprod_inl : forall {x y}, hom x (coprod x y);
  coprod_inr : forall {x y}, hom y (coprod x y);
  (* Destructor (Case Analysis / Co-Pairing): [f, g] *)
  coprod_map : forall {x y z} (f : hom x z) (g : hom y z), hom (coprod x y) z;
  (* Equations *)

  (* Beta rule: [f, g] o inl = f *)
  coprod_map_inl : forall {x y z} (f : hom x z) (g : hom y z), 
        compose (coprod_map f g) coprod_inl ≅ f;
  (* Beta rule: [f, g] o inr = g *)
  coprod_map_inr : forall {x y z} (f : hom x z) (g : hom y z), 
        compose (coprod_map f g) coprod_inr ≅ g;
  (* Eta rule: [h o inl, h o inr] = h *)
  coprod_eta : forall {x y z} (h : hom (coprod x y) z), 
        coprod_map (compose h coprod_inl) (compose h coprod_inr) ≅ h;
  (* Proper instance required for rewriting *)
  coprod_map_proper_ax : forall {x y z},
        Proper (hom_equiv ==> hom_equiv ==> hom_equiv) (@coprod_map x y z)
}.

Arguments initial {C} {Ct} {CC} {_}.
Arguments initial_map {C} {Ct} {CC} {_} {x}.
Arguments initial_map_unique {C} {Ct} {CC} {_} {x} _.

Arguments coprod {C} {Ct} {CC} {_} _ _.
Arguments coprod_inl {C} {Ct} {CC} {_} {x y}.
Arguments coprod_inr {C} {Ct} {CC} {_} {x y}.
Arguments coprod_map {C} {Ct} {CC} {_} {x y z} _ _.

Arguments coprod_map_inl {C} {Ct} {CC} {_} {x y z} _ _.
Arguments coprod_map_inr {C} {Ct} {CC} {_} {x y z} _ _.
Arguments coprod_eta {C} {Ct} {CC} {_} {x y z} _.
Arguments coprod_map_proper_ax {C} {Ct} {CC} {_} {x y z}.

(* Case analysis notation [f, g] *)
Notation "f '∐' g" := (coprod_map f g) (at level 30, no associativity) : type_scope.
(* Functorial map f + g (Torii dual) *)
Notation "f '⊔' g" := (coprod_map (compose coprod_inl f) (compose coprod_inr g)) 
                      (at level 30, no associativity) : type_scope.

(* --- Theorems --- *)

(* Uniqueness derived from Eta (mirroring prod_unique) *)
Theorem coprod_unique {C : SetoidCategory} {Ct : CartesianCategory C} {CC : CartesianClosedCategory C} {BCC : BiCartesianClosedCategory CC} : 
  forall {x y z} (f : hom x z) (g : hom y z) (h : hom (coprod x y) z), 
    (compose h coprod_inl ≅ f) /\ (compose h coprod_inr ≅ g) -> h ≅ (coprod_map f g).
Proof. Admitted.

(** * Derived Theorems *)

(* Directional version of coprod_unique *)
Theorem unique_coprod {C : SetoidCategory} {Ct : CartesianCategory C} {CC : CartesianClosedCategory C} {BCC : BiCartesianClosedCategory CC} : 
  forall (x y z : @obj C) (f1 : x → z) (f2 : y → z) (g : x ⊕ y → z), 
    ((g ∘ coprod_inl) ≅ f1) /\ ((g ∘ coprod_inr) ≅ f2) -> f1 ∐ f2 ≅ g.
Proof. Admitted.

(* Pre-composition distributes over coproduct map *)
(* Dual to compos_prod *)
Theorem compos_coprod {C : SetoidCategory} {Ct : CartesianCategory C} {CC : CartesianClosedCategory C} {BCC : BiCartesianClosedCategory CC} : 
  forall (x y z w : @obj C) (f : x → y) (g : z → y) (h : y → w),
 (h ∘ f) ∐ (h ∘ g) ≅ (h ∘ (f ∐ g)).
Proof. Admitted.

(* Identity expanded as a coproduct of injections *)
(* Dual to pr_and_id *)
Lemma in_and_id {C : SetoidCategory} {Ct : CartesianCategory C} {CC : CartesianClosedCategory C} {BCC : BiCartesianClosedCategory CC} : 
  forall (A B : @obj C), id (A ⊕ B) ≅ coprod_inl ∐ coprod_inr.
Proof. Admitted.

(* Functoriality of Coproduct (Composition of Parallel Maps) *)
(* Dual to prod_id *)
(* (g ⊔ id) ∘ (f ⊔ id) ≅ (g ∘ f) ⊔ id *)
Lemma coprod_id {C : SetoidCategory} {Ct : CartesianCategory C} {CC : CartesianClosedCategory C} {BCC : BiCartesianClosedCategory CC} 
  (X Y Z A : @obj C) : 
  forall (f : X → Y) (g : Y → Z),
    (g ⊔ (id A)) ∘ (f ⊔ (id A)) ≅ (g ∘ f) ⊔ (id A).
Proof. Admitted.

Lemma coprod_iso_cong {C : SetoidCategory} {Ct : CartesianCategory C} {CC : CartesianClosedCategory C} {BCC : BiCartesianClosedCategory CC} {A B A' B' : @obj C} :
  A ≅O A' -> B ≅O B' -> coprod A B ≅O coprod A' B'.
Proof. Admitted.

Class CartesianCategory (C : SetoidCategory) := mk_CartesianCategory {
  
  (* --- TERMINAL --- *)
  terminal : @obj C;
  terminal_map : forall {x}, hom x terminal;
  terminal_map_unique : forall {x} (f : hom x terminal), f ≅ terminal_map;

  (* --- PRODUCT --- *)
  prod : @obj C -> @obj C -> @obj C;
  prod_map : forall {x y z} (f : hom x y) (g : hom x z), hom x (prod y z);
  prod_fst : forall {x y}, hom (prod x y) x;
  prod_snd : forall {x y}, hom (prod x y) y;
  
  (* Equations *)
  prod_map_fst : forall {x y z} (f : hom x y) (g : hom x z), 
    compose prod_fst (prod_map f g) ≅ f;
  prod_map_snd : forall {x y z} (f : hom x y) (g : hom x z), 
    compose prod_snd (prod_map f g) ≅ g;
  prod_eta : forall {x y z} (h : hom x (prod y z)),
    prod_map (compose prod_fst h) (compose prod_snd h) ≅ h;

  (* Proper *)
  prod_map_proper_ax : forall {x y z},
    Proper (hom_equiv ==> hom_equiv ==> hom_equiv) (@prod_map x y z)
}.

Arguments terminal {C} {_}.
Arguments terminal_map {C} {_} {x}.
Arguments terminal_map_unique {C} {_} {x} _.

Arguments prod {C} {_} _ _.
Arguments prod_map {C} {_} {x y z} _ _.
Arguments prod_fst {C} {_} {x y}.
Arguments prod_snd {C} {_} {x y}.

Arguments prod_map_fst {C} {_} {x y z} _ _.
Arguments prod_map_snd {C} {_} {x y z} _ _.
Arguments prod_eta {C} {_} {x y z} _.
Arguments prod_map_proper_ax {C} {_} {x y z}.

Existing Instance prod_map_proper_ax.

Notation "x × y" := (prod x y) (at level 40, left associativity) : type_scope.
Notation "f '∏' g" := (prod_map f g) (at level 30, no associativity) : type_scope.
Notation "f 'Π' g" := (prod_map (compose f prod_fst) (compose g prod_snd)) (at level 30, no associativity) : type_scope.

(* --- Theorems --- *)

Lemma prod_unique {C : SetoidCategory} {CC : CartesianCategory C} : 
  forall {x y z} (f : hom x y) (g : hom x z) (h : hom x (prod y z)), 
    (compose prod_fst h ≅ f) /\ (compose prod_snd h ≅ g) -> h ≅ (prod_map f g).
Proof. Admitted.

Theorem unique_prod {C : SetoidCategory} {CC : CartesianCategory C} : 
  forall (x y z : @obj C) (f1 : x → y) (f2 : x → z) (g : x → y × z), 
    ((prod_fst ∘ g) ≅ f1) /\ ((prod_snd ∘ g) ≅ f2) -> f1 ∏ f2 ≅ g.
Proof. Admitted.

(* Composition distributes over product map *)
Theorem compos_prod {C : SetoidCategory} {CC : CartesianCategory C} : 
  forall x y z w (f : w → y) (g : w → z) (h : x → w),
  (f ∘ h) ∏ (g ∘ h) ≅ ((f ∏ g) ∘ h).
Proof. Admitted.

Lemma pr_and_id {C : SetoidCategory} {CC : CartesianCategory C} : 
  forall (A B : @obj C), id (A × B) ≅ prod_fst ∏ prod_snd.
Proof. Admitted.

(* EZ HIÁNYZOTT: Functoriality of Product *)
Lemma prod_id {C : SetoidCategory} {CC : CartesianCategory C} (X Y Z A : @obj C) : 
  forall (f : X → Y) (g : Y → Z),
    (g Π (id A)) ∘ (f Π (id A)) ≅ (g ∘ f) Π (id A).
Proof. Admitted.

  (prod_item : @obj C -> @obj C -> @obj C) (prod_map_item : forall {x y z} (f : hom x y) (g : hom x z), hom x (prod_item y z))
  (prod_fst_item : forall {x y}, hom (prod_item x y) x) (prod_snd_item : forall {x y}, hom (prod_item x y) y)
  : CartesianCategory C.
Proof. Admitted.

Lemma prod_iso_cong {C : SetoidCategory} {CC : CartesianCategory C} {A B A' B' : @obj C} :
  A ≅O A' -> B ≅O B' -> prod A B ≅O prod A' B'.
Proof. Admitted.

(* A Ct : CartesianCategory C paraméter jelzi, hogy erre épülünk *)
Class CartesianClosedCategory (C : SetoidCategory) {Ct : CartesianCategory C} := mk_CartesianClosedCategory {
  
  (* --- EXPONENTIAL OBJECT --- *)
  exp : @obj C -> @obj C -> @obj C;
  
  (* Constructor: Lambda g / curry g *)
  (* prod_map a CC-ből jön! *)
  exp_map : forall {x y z} (g : hom (prod x y) z), hom x (exp z y);
  
  (* Destructor: eval / apply *)
  eval : forall {y z}, hom (prod (exp z y) y) z;
  
  (* Equations *)
  
  (* Beta rule *)
  exp_beta : forall {x y z} (g : hom (prod x y) z),
    compose eval (prod_map (compose (exp_map g) prod_fst) (compose (id y) prod_snd)) ≅ g;
                      
  (* Eta rule *)
  exp_eta : forall {x y z} (h : hom x (exp z y)),
    exp_map (compose eval (prod_map (compose h prod_fst) (compose (id y) prod_snd))) ≅ h;

  (* Proper instance *)
  exp_map_proper_ax : forall {x y z},
    Proper (hom_equiv ==> hom_equiv) (@exp_map x y z)
}.

Require Import List. 
Import ListNotations.
Require Import Coq.Setoids.Setoid.
Require Import Coq.Classes.Morphisms.
Require Import Coq.Classes.RelationClasses.

Class SetoidCategory := mk_SetoidCategory {
  obj : Type;
  hom : obj -> obj -> Type;

  id : forall x, hom x x;
  compose : forall {x y z}, hom y z -> hom x y -> hom x z;

  hom_equiv : forall {x y}, hom x y -> hom x y -> Prop;
  
  hom_equiv_equivalence : forall {x y}, Equivalence (@hom_equiv x y);

  comp_assoc : forall x y z w (f : hom z w) (g : hom y z) (h : hom x y),
        hom_equiv (compose f (compose g h)) (compose (compose f g) h);

  id_right : forall x y (f : hom x y), hom_equiv (compose f (id x)) f;
  id_left  : forall x y (f : hom x y), hom_equiv (compose (id y) f) f;

  comp_proper : forall {x y z} (f g: hom y z) (h i : hom x y), 
        hom_equiv f g -> hom_equiv h i -> hom_equiv (compose f h) (compose g i);
}.

Notation "x → y" := (hom x y) (at level 90, right associativity) : type_scope.
Notation "f ∘ g" := (compose f g) (at level 40, left associativity) : type_scope.
Notation "f ≅ g" := (hom_equiv f g) (at level 40, no associativity) : type_scope.

Add Parametric Relation (C : SetoidCategory) (x y : @obj C) : (@hom C x y) (@hom_equiv C x y)
  reflexivity proved by (@Equivalence_Reflexive _ _ (@hom_equiv_equivalence C x y))
  symmetry proved by (@Equivalence_Symmetric _ _ (@hom_equiv_equivalence C x y))
  transitivity proved by (@Equivalence_Transitive _ _ (@hom_equiv_equivalence C x y))
  as hom_equiv_rel.

Global Instance compose_proper_instance {C : SetoidCategory} {x y z : @obj C} : 
  Proper (@hom_equiv C y z ==> @hom_equiv C x y ==> @hom_equiv C x z) (@compose C x y z).
Proof. Admitted.

(* Thin Category  *)
Class Thin (C : SetoidCategory) :=
  thin_unique : forall {x y} (f g : hom x y), f ≅ g.

(* Isomorphisms *)
Class Iso {C : SetoidCategory} (A B : @obj C) := {
  iso_to : A → B;
  iso_from : B → A;
  
  iso_to_from : iso_to ∘ iso_from ≅ id B;
  
  iso_from_to : iso_from ∘ iso_to ≅ id A
}.

Notation "A ≅O B" := (Iso A B) (at level 40, no associativity) : type_scope.

(* Tranzitivitás (A ≅O B és B ≅O X -> A ≅O X) *)

  (H1 : A ≅O B) (H2 : B ≅O X) : A ≅O X.

  Equivalence_Symmetric := iso_sym_inst;
  Equivalence_Transitive := iso_trans_inst

Class Functor (C D : SetoidCategory) := mk_Functor {
  (* Object: *)
  fobj : @obj C -> @obj D;

  (* Morphism: (x -> y) -> (F x -> F y) *)
  fmap : forall {x y : @obj C} (f : @hom C x y), @hom D (fobj x) (fobj y);

  (* Proper: *)
  fmap_proper : forall {x y : @obj C}, 
    Proper (@hom_equiv C x y ==> @hom_equiv D (fobj x) (fobj y)) (@fmap x y);

  (* Identity: F(id) ~ id *)
  fmap_id : forall (x : @obj C), 
    @hom_equiv D (fobj x) (fobj x) (fmap (@id C x)) (@id D (fobj x));

  (* Composition: F(g ∘ f) ~ F(g) ∘ F(f) *)
  fmap_comp : forall {x y z : @obj C} (f : @hom C x y) (g : @hom C y z),
    @hom_equiv D (fobj x) (fobj z) 
      (fmap (@compose C x y z g f)) 
      (@compose D (fobj x) (fobj y) (fobj z) (fmap g) (fmap f))
}.

Arguments fobj {C D} _ _.
Arguments fmap {C D} _ {x y} _.
Arguments fmap_proper {C D} _ {x y} _ _ _.
Arguments fmap_id {C D} _ _.
Arguments fmap_comp {C D} _ {x y z} _ _.

(* Proper instance for setoid_rewrite *)
Existing Instance fmap_proper.

Definition ComposeFunctors {C D E : SetoidCategory} (G : Functor D E) (F : Functor C D) : Functor C E.

Require Import Coq.Classes.Morphisms.

(** * Monads and Comonads *)

(** ** Definitions *)

Class Monad {C : SetoidCategory} (M : Functor C C) := mk_monad {
  return_op : forall {A}, @hom C A (@fobj C C M A);
  join_op : forall {A}, @hom C (@fobj C C M (@fobj C C M A)) (@fobj C C M A);

  monad_law_left_unit : forall {A},
    @compose C (@fobj C C M A) (@fobj C C M (@fobj C C M A)) (@fobj C C M A)
      (@join_op A) 
      (@fmap C C M A (@fobj C C M A) (@return_op A)) 
    ≅ @id C (@fobj C C M A);

  monad_law_right_unit : forall {A},
    @compose C (@fobj C C M A) (@fobj C C M (@fobj C C M A)) (@fobj C C M A)
      (@join_op A) 
      (@return_op (@fobj C C M A)) 
    ≅ @id C (@fobj C C M A);

  monad_law_assoc : forall {A},
    @compose C (@fobj C C M (@fobj C C M (@fobj C C M A))) (@fobj C C M (@fobj C C M A)) (@fobj C C M A)
      (@join_op A) 
      (@fmap C C M _ _ (@join_op A)) 
    ≅ 
    @compose C (@fobj C C M (@fobj C C M (@fobj C C M A))) (@fobj C C M (@fobj C C M A)) (@fobj C C M A)
      (@join_op A)
      (@join_op (@fobj C C M A))
}.

Class Comonad {C : SetoidCategory} (W : Functor C C) := mk_comonad {
  extract : forall {A}, @hom C (@fobj C C W A) A;
  duplicate : forall {A}, @hom C (@fobj C C W A) (@fobj C C W (@fobj C C W A));

  comonad_law_left_unit : forall {A},
    @compose C (@fobj C C W A) (@fobj C C W (@fobj C C W A)) (@fobj C C W A)
      (@fmap C C W _ _ (@extract A)) 
      (@duplicate A) 
    ≅ @id C (@fobj C C W A);

  comonad_law_right_unit : forall {A},
    @compose C (@fobj C C W A) (@fobj C C W (@fobj C C W A)) (@fobj C C W A)
      (@extract (@fobj C C W A)) 
      (@duplicate A) 
    ≅ @id C (@fobj C C W A);

  comonad_law_assoc : forall {A},
    @compose C (@fobj C C W A) (@fobj C C W (@fobj C C W A)) (@fobj C C W (@fobj C C W (@fobj C C W A)))
      (@fmap C C W _ _ (@duplicate A)) 
      (@duplicate A) 
    ≅ 
    @compose C (@fobj C C W A) (@fobj C C W (@fobj C C W A)) (@fobj C C W (@fobj C C W (@fobj C C W A)))
      (@duplicate (@fobj C C W A)) 
      (@duplicate A)
}.

(** ** Bind/Extend Style definitions *)

Class Monad_Bind_Style {Cat : SetoidCategory} (M : Functor Cat Cat) := mk_Monad_Bind_Style {
  return_b : forall {A}, @hom Cat A (@fobj Cat Cat M A);
  bind_b : forall {A B}, (@hom Cat A (@fobj Cat Cat M B)) -> (@hom Cat (@fobj Cat Cat M A) (@fobj Cat Cat M B));
  
  bind_law1 : forall {A B} (f : @hom Cat A (@fobj Cat Cat M B)), 
    (@compose Cat A (@fobj Cat Cat M A) (@fobj Cat Cat M B) (bind_b f) return_b) ≅ f;
    
  bind_law2 : forall {A}, bind_b (@return_b A) ≅ @id Cat (@fobj Cat Cat M A);
  
  bind_law3 : forall {A B C} (f : @hom Cat A (@fobj Cat Cat M B)) (g : @hom Cat B (@fobj Cat Cat M C)),
    @compose Cat (@fobj Cat Cat M A) (@fobj Cat Cat M B) (@fobj Cat Cat M C) (bind_b g) (bind_b f) 
    ≅ bind_b (@compose Cat A (@fobj Cat Cat M B) (@fobj Cat Cat M C) (bind_b g) f)
}.

Class CoMonad_Extend_Style {Cat : SetoidCategory} (W : Functor Cat Cat) := mk_CoMonad_Extend_Style {
  extract_b : forall {A}, @hom Cat (@fobj Cat Cat W A) A;
  extend_b : forall {A B}, (@hom Cat (@fobj Cat Cat W B) A) -> (@hom Cat (@fobj Cat Cat W B) (@fobj Cat Cat W A));
  
  extend_law1: forall {A B} (f : @hom Cat (@fobj Cat Cat W B) A), 
    (@compose Cat (@fobj Cat Cat W B) (@fobj Cat Cat W A) A extract_b (extend_b f)) ≅ f;
    
  extend_law2: forall {A}, extend_b (@extract_b A) ≅ @id Cat (@fobj Cat Cat W A);
  
  extend_law3: forall {A B C} (f : @hom Cat (@fobj Cat Cat W C) B) (g : @hom Cat (@fobj Cat Cat W B) A),
    @compose Cat (@fobj Cat Cat W C) (@fobj Cat Cat W B) (@fobj Cat Cat W A) (extend_b g) (extend_b f) 
    ≅ extend_b (@compose Cat (@fobj Cat Cat W C) (@fobj Cat Cat W B) A g (extend_b f))
}.

(** ** Conversions *)

Definition bind_from_join {Cat : SetoidCategory} {M : Functor Cat Cat} {MJ : Monad M} {A B}
  (f : @hom Cat A (@fobj Cat Cat M B)) : @hom Cat (@fobj Cat Cat M A) (@fobj Cat Cat M B) :=
  @compose Cat (@fobj Cat Cat M A) (@fobj Cat Cat M (@fobj Cat Cat M B)) (@fobj Cat Cat M B)
    (@join_op Cat M MJ B) 
    (@fmap Cat Cat M A (@fobj Cat Cat M B) f).

Definition join_from_bind {Cat : SetoidCategory} {M : Functor Cat Cat}  {MB : Monad_Bind_Style M} {A}
  : @hom Cat (@fobj Cat Cat M (@fobj Cat Cat M A)) (@fobj Cat Cat M A) :=
  @bind_b Cat M MB (@fobj Cat Cat M A) A (@id Cat (@fobj Cat Cat M A)).

(** ** Adjunctions generate Monads and Comonads *)

(* Every Adjunction G -| F (where G:C->D is Right Adjoint) induces a Monad on D.
   T = G o F
*)
Instance AdjFuncMonad (C D : SetoidCategory) (G : Functor C D) (RA : IsRightAdjoint G): 
  Monad_Bind_Style (ComposeFunctors (C:=D) (D:=C) (E:=D) G (LeftAdjointFunctor G RA)).
Proof. Admitted.

(* Every Adjunction F -| G (where F:D->C is Left Adjoint) induces a Comonad on C.
   W = F o G
*)
Instance AdjFuncCoMonad (C D : SetoidCategory) (F : Functor D C) (LA : IsLeftAdjoint F): 
  CoMonad_Extend_Style (ComposeFunctors (C:=C) (D:=D) (E:=C) F (RightAdjointFunctor F LA)).
Proof. Admitted.
